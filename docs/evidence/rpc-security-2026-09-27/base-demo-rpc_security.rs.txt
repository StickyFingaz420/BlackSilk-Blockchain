#[derive(Default)]
struct CountingPow(AtomicUsize);
impl PowFunction for CountingPow {
    fn pow_hash(&self, _: &Hash, _: &[u8]) -> Hash {
        self.0.fetch_add(1, Ordering::SeqCst);
        [0; 32]
    }
}

fn serve() -> (tokio::runtime::Runtime, SocketAddr, Shared, Arc<CountingPow>) {
    let params = ChainParams::regtest();
    let rules = TxRules::for_chain(&params);
    let pow = Arc::new(CountingPow::default());
    let m = ChainManager::open(params, rules, pow.clone(), Box::<MemoryStore>::default(), [3; 32])
        .unwrap();
    let shared: Shared = Arc::new(Mutex::new(m));
    let rt = tokio::runtime::Runtime::new().unwrap();
    let l = rt.block_on(tokio::net::TcpListener::bind("127.0.0.1:0")).unwrap();
    let addr = l.local_addr().unwrap();
    let app = router(shared.clone());
    rt.spawn(async move { axum::serve(l, app).await.unwrap() });
    (rt, addr, shared, pow)
}

fn status(addr: SocketAddr, raw: &[u8]) -> Option<u16> {
    let mut s = TcpStream::connect(addr).ok()?;
    s.set_read_timeout(Some(Duration::from_secs(10))).ok()?;
    s.write_all(raw).ok()?;
    let mut resp = Vec::new();
    let _ = s.read_to_end(&mut resp);
    let resp = String::from_utf8_lossy(&resp).into_owned();
    resp.split(' ').nth(1)?.parse().ok()
}

fn block_on(m: &ChainManager, parent: &Hash, seed: u64) -> Block {
    let t = m.template_on(parent).unwrap();
    let mut rng = ChaCha20Rng::seed_from_u64(seed);
    let (miner, _) = WalletKeys::generate(&mut rng);
    let cb = build_coinbase(
        t.height,
        &[Payment { address: miner.address(SubaddressIndex::PRIMARY), amount: t.reward }],
        &miner.hedge_secret(),
        &mut rng,
    )
    .unwrap();
    let txs = vec![Transaction::Coinbase(cb)];
    let ids: Vec<Hash> = txs.iter().map(Transaction::hash).collect();
    let header = BlockHeader {
        version: t.version,
        height: t.height,
        prev_id: t.prev_id,
        timestamp: t.min_timestamp.max(m.params().genesis.timestamp + 120 * t.height),
        difficulty: t.difficulty,
        tx_root: tx_root(&ids),
        nonce: seed,
    };
    Block { header, txs }
}

#[test]
fn demo_rebinding_host_is_refused() {
    let (_rt, addr, ..) = serve();
    let s = status(addr, b"GET /info HTTP/1.1\r\nHost: evil.example:29333\r\nConnection: close\r\n\r\n");
    assert_eq!(s, Some(403), "a DNS-rebinding Host was served");
}

#[test]
fn demo_cross_site_request_is_refused() {
    let (_rt, addr, ..) = serve();
    let req = format!(
        "GET /blocks?from=0&count=100 HTTP/1.1\r\nHost: {addr}\r\nSec-Fetch-Site: cross-site\r\nSec-Fetch-Mode: no-cors\r\nConnection: close\r\n\r\n"
    );
    assert_eq!(status(addr, req.as_bytes()), Some(403), "a cross-site no-cors GET was served");
}

#[test]
fn demo_unauthenticated_info_is_refused() {
    let (_rt, addr, ..) = serve();
    let req = format!("GET /info HTTP/1.1\r\nHost: {addr}\r\nConnection: close\r\n\r\n");
    assert_eq!(status(addr, req.as_bytes()), Some(401), "/info served without a credential");

#[test]
fn demo_oversized_tx_body_is_refused() {
    let (_rt, addr, ..) = serve();
    let body = format!("{{\"hex\":\"{}\"}}", "00".repeat(6 * 1024 * 1024));
    let req = format!(
        "POST /tx HTTP/1.1\r\nHost: {addr}\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{body}",
        body.len()
    );
    assert_eq!(status(addr, req.as_bytes()), Some(413), "a 12 MiB /tx body was read and parsed");
}

#[test]
fn demo_oversized_headers_are_refused() {
    let (_rt, addr, ..) = serve();
    let req = format!(
        "GET /info HTTP/1.1\r\nHost: {addr}\r\nX-Pad: {}\r\nConnection: close\r\n\r\n",
        "a".repeat(100 * 1024)
    );
    assert_eq!(status(addr, req.as_bytes()), Some(431), "100 KiB of headers accepted");
}

#[test]
fn demo_slow_headers_are_cut() {
    let (_rt, addr, ..) = serve();
    let mut s = TcpStream::connect(addr).unwrap();
    s.write_all(b"GET /info HTTP/1.1\r\nHost: 127.0.0.1\r\n").unwrap();
    std::thread::sleep(Duration::from_secs(12));
    s.set_read_timeout(Some(Duration::from_secs(1))).unwrap();
    let mut buf = [0u8; 16];
    let closed = matches!(s.read(&mut buf), Ok(0));
    assert!(closed, "a partial request head was kept open for 12 s");
}

#[test]
fn demo_connection_flood_is_capped() {
    let (_rt, addr, ..) = serve();
    let mut open = Vec::new();
    for _ in 0..200 {
        open.push(TcpStream::connect(addr).unwrap());
    }
    // Every connection is kept: none is closed by the server.
    std::thread::sleep(Duration::from_millis(500));
    let mut closed = 0;
    for s in &mut open {
        s.set_read_timeout(Some(Duration::from_millis(20))).unwrap();
        let mut b = [0u8; 1];
        if matches!(s.read(&mut b), Ok(0)) {
            closed += 1;
        }
    }
    assert!(closed > 0, "200 idle connections all held open");
}

#[test]
fn demo_deep_fork_block_costs_no_pow() {
    let (_rt, addr, shared, pow) = serve();
    let parent = {
        let mut m = shared.lock().unwrap();
        for i in 0..20 {
            let tip = m.tip_id();
            let b = block_on(&m, &tip, 100 + i);
            let now = b.header.timestamp;
            m.submit_block(b, now).unwrap();
        }
        m.headers().main_id_at(2).unwrap()
    };
    let fork = block_on(&shared.lock().unwrap(), &parent, 7);
    let fork_id = fork.header.id(ChainParams::regtest().network_id);
    let before = pow.0.load(Ordering::SeqCst);
    let body = format!("{{\"hex\":\"{}\"}}", hex::encode(fork.encode()));
    let req = format!(
        "POST /block HTTP/1.1\r\nHost: {addr}\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{body}",
        body.len()
    );
    assert_eq!(status(addr, req.as_bytes()), Some(200));
    let calls = pow.0.load(Ordering::SeqCst) - before;
    let stored = shared.lock().unwrap().header(&fork_id).is_some();
    assert_eq!((calls, stored), (0, false), "a deep-fork block cost {calls} PoW calls; stored: {stored}");
}
